class TrajectoryPlanner : public rclcpp::Node

public:
  TrajectoryPlanner() : Node("trajectory_planner")
  { // Parameters
    this->declare_parameter("waypoints_file", "");
    this->declare_parameter("waypoint_radius", 2.2);  // Acceptance radius in meters
    this->declare_parameter("hover_time", 2.0);       // Time to hover at each waypoint
    this->declare_parameter("loop_trajectory", false);
    this->declare_parameter("flight_speed", 1.0); // m/s

    waypoint_radius_ = this->get_parameter("waypoint_radius").as_double();
    hover_time_ = this->get_parameter("hover_time").as_double();
    loop_trajectory_ = this->get_parameter("loop_trajectory").as_bool();
    flight_speed_ = this->get_parameter("flight_speed").as_double();
    
    std::string waypoints_file = this->get_parameter("waypoints_file").as_string();

    // QoS for PX4
    auto qos = rclcpp::QoS(rclcpp::KeepLast(10));
    qos.best_effort();
    qos.transient_local();

    // Publisher
    pub_setpoint_ = create_publisher<geometry_msgs::msg::PointStamped>(
      "/control/position_setpoint", 10);

    pub_arm_command_ = create_publisher<std_msgs::msg::Float64MultiArray>(
      "/arm_position_controller/commands", 10);

    // Subscriber
    sub_position_ = create_subscription<px4_msgs::msg::VehicleLocalPosition>(
      "/fmu/out/vehicle_local_position_v1", qos,
      std::bind(&TrajectoryPlanner::positionCallback, this, std::placeholders::_1));

    // Timer for setpoint publishing
    timer_ = create_wall_timer(
      std::chrono::milliseconds(100),  // 10 Hz
      std::bind(&TrajectoryPlanner::publishSetpoint, this));

    // Load waypoints
    if (waypoints_file.empty()) {
      loadDefaultWaypoints();
    } else {
      loadWaypointsFromFile(waypoints_file);
    }

    if (!waypoints_.empty()) {
        current_setpoint_ = waypoints_[0];
        last_wp_ = waypoints_[0];
        target_wp_ = waypoints_[0];
    }

    printWaypoints();
    
    RCLCPP_INFO(get_logger(), "═══════════════════════════════════════════");
    RCLCPP_INFO(get_logger(), "  Trajectory Planner Started");
    RCLCPP_INFO(get_logger(), "  Waypoints: %zu", waypoints_.size());
    RCLCPP_INFO(get_logger(), "  Acceptance radius: %.2f m", waypoint_radius_);
    RCLCPP_INFO(get_logger(), "  Loop: %s", loop_trajectory_ ? "Yes" : "No");
    RCLCPP_INFO(get_logger(), "═══════════════════════════════════════════");
  }

  void reportPerformance(bool interrupted = false)
  {
    if (report_printed_) return;
    double rmse = (sample_count_ > 0) ? std::sqrt(sum_sq_error_pos_ / (double)sample_count_) : 0.0;
    
    RCLCPP_INFO(get_logger(), "═══════════════════════════════════════════");
    RCLCPP_INFO(get_logger(), "  %s", interrupted ? "SIMULATION STOPPED" : "TRAJECTORY COMPLETE");
    RCLCPP_INFO(get_logger(), "═══════════════════════════════════════════");
    RCLCPP_INFO(get_logger(), "  Total Samples: %d", sample_count_);
    RCLCPP_INFO(get_logger(), "  Max Pos Error: %.4f m", max_error_pos_);
    RCLCPP_INFO(get_logger(), "  RMSE Position: %.4f m", rmse);
    RCLCPP_INFO(get_logger(), "═══════════════════════════════════════════");
    report_printed_ = true;
  }

  void loadWaypointsFromFile(const std::string& filename)
  {
    try {
      RCLCPP_INFO(get_logger(), "Loading waypoints from: %s", filename.c_str());
      YAML::Node config = YAML::LoadFile(filename);
      
      if (!config["waypoints"]) {
        RCLCPP_ERROR(get_logger(), "No 'waypoints' key in YAML file!");
        loadDefaultWaypoints();
        return;
      }

      waypoints_.clear();
      for (const auto& wp : config["waypoints"]) {
        Waypoint w;
        w.x = wp["x"].as<double>();
        w.y = wp["y"].as<double>();
        w.z = wp["z"].as<double>();
        w.name = wp["name"] ? wp["name"].as<std::string>() : "";
        waypoints_.push_back(w);
      }

      if (waypoints_.empty()) {
        RCLCPP_ERROR(get_logger(), "No waypoints loaded!");
        loadDefaultWaypoints();
      }
    } catch (const std::exception& e) {
      RCLCPP_ERROR(get_logger(), "Failed to load waypoints: %s", e.what());
      loadDefaultWaypoints();
    }
  }

  void printWaypoints()
  {
    RCLCPP_INFO(get_logger(), "Trajectory waypoints:");
    for (size_t i = 0; i < waypoints_.size(); i++) {
      const auto& wp = waypoints_[i];
      RCLCPP_INFO(get_logger(), "  [%zu] %-15s → (%.2f, %.2f, %.2f)", 
                  i, wp.name.c_str(), wp.x, wp.y, wp.z);
    }
  }

  void positionCallback(const px4_msgs::msg::VehicleLocalPosition::SharedPtr msg)

  void advanceToNextWaypoint()
  {
    hovering_ = false;
    current_waypoint_idx_++;

    if (current_waypoint_idx_ >= waypoints_.size()) {
      if (loop_trajectory_) {
        RCLCPP_INFO(get_logger(), "↻ Looping back to start");
        current_waypoint_idx_ = 0;
      } else {
        RCLCPP_INFO(get_logger(), "✓ Trajectory complete!");
        trajectory_complete_ = true;
        reportPerformance();
      }
    } else {
      const auto& next_wp = waypoints_[current_waypoint_idx_];
      RCLCPP_INFO(get_logger(), "→ Next waypoint [%zu]: %s", 
                  current_waypoint_idx_, next_wp.name.c_str());
    }
  }

    void publishSetpoint()
  {
    if (waypoints_.empty() || trajectory_complete_ || !position_received_|| !metrics_started_) {
      return;
    }

    // --- 1. INTERPOLATE SETPOINT ---
    if (!hovering_) {
        // Calculate distance between last WP and Target WP
        double segment_dist = calculateDistance(last_wp_.x, last_wp_.y, last_wp_.z, 
                                                target_wp_.x, target_wp_.y, target_wp_.z);
        
        if (segment_dist > 0.001) {
            // Move progress based on speed (dt = 0.02s)
            double step = (flight_speed_ * 0.02) / segment_dist;
            interpolation_progress_ += step;
        } else {
            interpolation_progress_ = 1.0;
        }

        if (interpolation_progress_ >= 1.0) {
            interpolation_progress_ = 1.0;
            // We have conceptually reached the setpoint, now wait for drone to actually get there
            // But for setpoint generation, we hold at target
        }

        // Linear Interpolation (Lerp)
        current_setpoint_.x = last_wp_.x + (target_wp_.x - last_wp_.x) * interpolation_progress_;
        current_setpoint_.y = last_wp_.y + (target_wp_.y - last_wp_.y) * interpolation_progress_;
        current_setpoint_.z = last_wp_.z + (target_wp_.z - last_wp_.z) * interpolation_progress_;
    }

    // --- 2. CHECK ARRIVAL (Actual Drone vs Target WP) ---
    double dist_to_target = calculateDistance(current_position_[0], current_position_[1], current_position_[2], 
                                              target_wp_.x, target_wp_.y, target_wp_.z);

    if (interpolation_progress_ >= 1.0 && dist_to_target < waypoint_radius_ && !hovering_) {
        RCLCPP_INFO(get_logger(), "✓ Reached [%s]", target_wp_.name.c_str());
        hovering_ = true;
        hover_start_time_ = this->now();
    }

    // --- 3. HANDLE HOVER & SWITCHING ---
    if (hovering_) {
        // Hold setpoint at target
        current_setpoint_ = target_wp_; 

        if ((this->now() - hover_start_time_).seconds() >= hover_time_) {
            hovering_ = false;
            current_waypoint_idx_++;
            
            if (current_waypoint_idx_ >= waypoints_.size()) {
                if (loop_trajectory_) {
                    // Loop back to start (be careful with interpolation here)
                    last_wp_ = target_wp_;
                    target_wp_ = waypoints_[0];
                    current_waypoint_idx_ = 0;
                    interpolation_progress_ = 0.0;
                } else {
                    reportPerformance(false);
                    trajectory_complete_ = true;
                }
            } else {
                // Setup next leg
                last_wp_ = target_wp_;
                target_wp_ = waypoints_[current_waypoint_idx_];
                interpolation_progress_ = 0.0;
                RCLCPP_INFO(get_logger(), "→ Moving to [%s]", target_wp_.name.c_str());
            }
        }
    }

    // --- 4. PUBLISH SETPOINT ---
    auto msg = geometry_msgs::msg::PointStamped();
    msg.header.stamp = this->now();
    msg.header.frame_id = "map";
    msg.point.x = current_setpoint_.x;
    msg.point.y = current_setpoint_.y;
    msg.point.z = current_setpoint_.z;
    pub_setpoint_->publish(msg);

    // --- 5. CALCULATE METRICS (Drone vs Current Interpolated Setpoint) ---
    double err = calculateDistance(current_position_[0], current_position_[1], current_position_[2],
                                   current_setpoint_.x, current_setpoint_.y, current_setpoint_.z);
    
    sum_sq_error_pos_ += err * err;
    if (err > max_error_pos_) max_error_pos_ = err;
    sample_count_++;

    // --- 6. ARM COMMAND ---
    updateArmTrajectory();
  }


  void updateArmTrajectory()
  {
    std::vector<double> joints = {-1.57, 3.14, 0.0}; // Default safe pose

    if (arm_active_) {
        double t = (this->now() - arm_start_time_).seconds();
        
        // --- Oscillation Logic ---
        // Center: -1.57 (Straight Down)
        // Amplitude: 1.0 rad (~57 degrees) -> Swing from -0.57 to -2.57
        // Frequency: 0.5 Hz (2 seconds per cycle)
        double amplitude = 1.0; 
        double amplitude_2 = 1.3; 

        double frequency = 0.1; 
        
        // J1: Sine Wave
        joints[0] = -1.57 + amplitude * std::sin(2.0 * M_PI * frequency * t);
        
        // J2: 0.0 (Fully Extended) - Maximizes disturbance torque
        joints[1] = 0.0 + amplitude_2 * std::sin(2.0 * M_PI * frequency * t);
        
        // J3: 0.0 (Unused)
        joints[2] = 0.0;
    }

    std_msgs::msg::Float64MultiArray arm_msg;
    arm_msg.data = joints;
    pub_arm_command_->publish(arm_msg);
  }

  double calculateDistance(double x1, double y1, double z1, 
                          double x2, double y2, double z2)
  {
    return std::sqrt(std::pow(x2 - x1, 2) + 
                    std::pow(y2 - y1, 2) + 
                    std::pow(z2 - z1, 2));
  }

  // Publishers & Subscribers
  rclcpp::Publisher<geometry_msgs::msg::PointStamped>::SharedPtr pub_setpoint_;
  rclcpp::Publisher<std_msgs::msg::Float64MultiArray>::SharedPtr pub_arm_command_;
  rclcpp::Subscription<px4_msgs::msg::VehicleLocalPosition>::SharedPtr sub_position_;
  rclcpp::TimerBase::SharedPtr timer_;

  // Waypoints
  std::vector<Waypoint> waypoints_;
  Waypoint current_setpoint_, last_wp_, target_wp_;
  size_t current_waypoint_idx_ = 0;
  
  // State
  double current_position_[3] = {0.0, 0.0, 0.0};
  bool position_received_ = false;
  bool hovering_ = false;
  bool trajectory_complete_ = false;
  bool arm_active_ = false;

  rclcpp::Time hover_start_time_{0, 0, RCL_ROS_TIME};
  rclcpp::Time arm_start_time_{0, 0, RCL_ROS_TIME};
  
  // Parameters
  double waypoint_radius_;
  double hover_time_;
  double flight_speed_;
  bool loop_trajectory_;
  double interpolation_progress_ = 0.0;
  int log_counter_ = 0;
};

int main(int argc, char** argv)
{
  rclcpp::init(argc, argv);
  auto node = std::make_shared<TrajectoryPlanner>();
  rclcpp::spin(node);
  node->reportPerformance(true);
  rclcpp::shutdown();
  return 0;
}
